\section{Simulations}


\subsection{Background on Unit Quaternions}
Let $q\in S^3$ be a unit quaternion, given by
\begin{equation}
    q = q_w + q_x i + q_y j + q_z k  = q_r + q_v\in S^3.
\end{equation}

The conjugate of $q$ is denoted $q^*$ and given by
\begin{equation}
    q^* = q_w - q_x i - q_y j - q_z k = q_r - q_v.
\end{equation}
It holds that $(q^*)^* = q$, and $(pq)^* = q^* p^*$.

The quaternion product is given by
\begin{align}
    pq 
    &= (p_r q_r - p_v \cdot q_v) + (p_r q_v + q_r p_v + p_v \times q_v)
    \\
    &= (p\cdot q^*) + (p_r q_v + q_r p_v + p_v \times q_v).
\end{align}

A rotation of $\theta$ radians around the axis $u\in\R{3}$ is given by
\begin{equation}
    q = \cos \frac{\theta}{2} + u \sin \frac{\theta}{2}.
\end{equation}

Represent a vector $v_v\in\R{3}$ as a pure quaternion, $v=(0,v_v).$
The rotation of a vector $v_v\in\R{3}$ by a unit quaternion $q$ is given by
\begin{equation}
    L_q(v) = q v q^*.
\end{equation}

It follows that the composition of $L_{q2}$ and $L_{q1}$ is given by
\begin{equation}
    L_{q2}(L_{q1}(v)) = L_{q2 \, q1}(v).
\end{equation}

Equivalently, the rotation of a vector by a quaternion can written using rotation matrices.
Let $R(q) \in SO(3)$ be the rotation matrix induced by $q$, such that
\begin{equation}
    L_q(v) = R(q) v_v,
\end{equation}
where
\begin{equation}
    R(q) = \begin{bmatrix}
        1-2(q_y^2 + q_z^2) & 2(q_x q_y - q_z q_w) & 2(q_x q_z + q_y q_w) \\
        2(q_x q_y + q_z q_w) & 1-2(q_x^2 + q_z^2) & 2(q_y q_z - q_x q_w) \\
        2(q_x q_z - q_y q_w) & 2(q_y q_z + q_x q_w) & 1-2(q_x^2 + q_y^2)
    \end{bmatrix}.
\end{equation}

This representation can be differentiated with respect to $q_w$, $q_x$, $q_y$, and $q_z$, via
\begin{align}
    \frac{\partial R(q)}{\partial q_w}
    &= \begin{bmatrix}
        0 & -2q_z & 2q_y \\
        2q_z & 0 & -2q_x \\
        -2q_y & 2q_x & 0
    \end{bmatrix},
    \\
    \frac{\partial R(q)}{\partial q_x}
    &= \begin{bmatrix}
        0 & 2q_y & 2q_z \\
        2q_y & -4 q_x & -2q_w \\
        2q_z & 2q_w & -4q_x
    \end{bmatrix},
    \\
    \frac{\partial R(q)}{\partial q_y}
    &= \begin{bmatrix}
        -4q_y & 2q_x & 2q_w \\
        2q_x & 0 & 2q_z \\
        -2q_w & 2q_z & -4q_y
    \end{bmatrix},
    \\
    \frac{\partial R(q)}{\partial q_z}
    &= \begin{bmatrix}
        -4q_z & -2q_w & 2q_x \\
        2q_w & -4q_z & 2q_y \\
        2q_x & 2q_y & 0
    \end{bmatrix}.
\end{align}

Let $f$ be some function of $R(q)$, and let $\nabla_{R(q)} f$ denote the gradient of $f$ with respect to $R(q)$.
Then, the gradient of $f$ with respect to $q$ is given by
\begin{equation}
    \nabla_q f = \begin{bmatrix}
        \langle \nabla_{R(q)} f, \frac{\partial R(q)}{\partial q_w} \rangle_F
        \\
        \langle \nabla_{R(q)} f, \frac{\partial R(q)}{\partial q_x} \rangle_F
        \\
        \langle \nabla_{R(q)} f, \frac{\partial R(q)}{\partial q_y} \rangle_F
        \\
        \langle \nabla_{R(q)} f, \frac{\partial R(q)}{\partial q_z} \rangle_F
    \end{bmatrix}.
\end{equation}

Let $c\in\{w,x,y,z\}$.
For clarity, we can see
\begin{align}
    \left\langle \nabla_{R(q)} f, \frac{\partial R(q)}{\partial q_c} \right\rangle_F
    &= \left\langle \begin{bmatrix}
        \frac{\partial f}{\partial R(q)_{11}} & \frac{\partial f}{\partial R(q)_{12}} & \frac{\partial f}{\partial R(q)_{13}} \\
        \frac{\partial f}{\partial R(q)_{21}} & \frac{\partial f}{\partial R(q)_{22}} & \frac{\partial f}{\partial R(q)_{23}} \\
        \frac{\partial f}{\partial R(q)_{31}} & \frac{\partial f}{\partial R(q)_{32}} & \frac{\partial f}{\partial R(q)_{33}} \\
    \end{bmatrix}, \begin{bmatrix}
        \frac{\partial R(q)_{11}}{\partial q_c} & \frac{\partial R(q)_{12}}{\partial q_c} & \frac{\partial R(q)_{13}}{\partial q_c} \\
        \frac{\partial R(q)_{21}}{\partial q_c} & \frac{\partial R(q)_{22}}{\partial q_c} & \frac{\partial R(q)_{33}}{\partial q_c} \\
        \frac{\partial R(q)_{31}}{\partial q_c} & \frac{\partial R(q)_{32}}{\partial q_c} & \frac{\partial R(q)_{33}}{\partial q_c} \\
    \end{bmatrix} \right\rangle_F
    \\
    &= \trace\left( \begin{bmatrix}
        \frac{\partial f}{\partial R(q)_{11}} & \frac{\partial f}{\partial R(q)_{12}} & \frac{\partial f}{\partial R(q)_{13}} \\
        \frac{\partial f}{\partial R(q)_{21}} & \frac{\partial f}{\partial R(q)_{22}} & \frac{\partial f}{\partial R(q)_{23}} \\
        \frac{\partial f}{\partial R(q)_{31}} & \frac{\partial f}{\partial R(q)_{32}} & \frac{\partial f}{\partial R(q)_{33}} \\
    \end{bmatrix}^\top \begin{bmatrix}
        \frac{\partial R(q)_{11}}{\partial q_c} & \frac{\partial R(q)_{12}}{\partial q_c} & \frac{\partial R(q)_{13}}{\partial q_c} \\
        \frac{\partial R(q)_{21}}{\partial q_c} & \frac{\partial R(q)_{22}}{\partial q_c} & \frac{\partial R(q)_{33}}{\partial q_c} \\
        \frac{\partial R(q)_{31}}{\partial q_c} & \frac{\partial R(q)_{32}}{\partial q_c} & \frac{\partial R(q)_{33}}{\partial q_c} \\
    \end{bmatrix} \right)
    \\
    &= \trace\left( \begin{bmatrix}
        \frac{\partial f}{\partial R(q)_{11}} & \frac{\partial f}{\partial R(q)_{21}} & \frac{\partial f}{\partial R(q)_{31}} \\
        \frac{\partial f}{\partial R(q)_{12}} & \frac{\partial f}{\partial R(q)_{22}} & \frac{\partial f}{\partial R(q)_{32}} \\
        \frac{\partial f}{\partial R(q)_{13}} & \frac{\partial f}{\partial R(q)_{23}} & \frac{\partial f}{\partial R(q)_{33}} \\
    \end{bmatrix} \begin{bmatrix}
        \frac{\partial R(q)_{11}}{\partial q_c} & \frac{\partial R(q)_{12}}{\partial q_c} & \frac{\partial R(q)_{13}}{\partial q_c} \\
        \frac{\partial R(q)_{21}}{\partial q_c} & \frac{\partial R(q)_{22}}{\partial q_c} & \frac{\partial R(q)_{33}}{\partial q_c} \\
        \frac{\partial R(q)_{31}}{\partial q_c} & \frac{\partial R(q)_{32}}{\partial q_c} & \frac{\partial R(q)_{33}}{\partial q_c} \\
    \end{bmatrix} \right)
    \\
    &= \frac{\partial f}{\partial R(q)_{11}} \frac{\partial R(q)_{11}}{\partial q_c}
    + \frac{\partial f}{\partial R(q)_{21}} \frac{\partial R(q)_{21}}{\partial q_c}
    + \frac{\partial f}{\partial R(q)_{31}} \frac{\partial R(q)_{31}}{\partial q_c}
    \\&\qquad + \frac{\partial f}{\partial R(q)_{12}} \frac{\partial R(q)_{12}}{\partial q_c}
    + \frac{\partial f}{\partial R(q)_{22}} \frac{\partial R(q)_{22}}{\partial q_c}
    + \frac{\partial f}{\partial R(q)_{32}} \frac{\partial R(q)_{32}}{\partial q_c}
    \\&\qquad + \frac{\partial f}{\partial R(q)_{13}} \frac{\partial R(q)_{13}}{\partial q_c}
    + \frac{\partial f}{\partial R(q)_{23}} \frac{\partial R(q)_{23}}{\partial q_c}
    + \frac{\partial f}{\partial R(q)_{33}} \frac{\partial R(q)_{33}}{\partial q_c}
    \\
    &= \frac{\partial f}{\partial q_c}.
\end{align}


\subsection{Toy Euclidean Problem}
Manifold: 3-dimensional Euclidean space,
\begin{equation}
    \mathcal{M}_i = \R{3}.
\end{equation}

Objective function:
\begin{equation}
    f(\mathcal{D}) = \sum_{(i,j)\in E} \norm{x_j - x_i - \tilde{d}_{ij}}_{R_{ij}^{-1}}^2,
\end{equation}
where $E\in\seq{N}^2$ is a set of directed measurement edges and $\tilde{d}_{ij}$ is a noisy measurement given by
\begin{equation}
    \tilde{d}_{ij} = x_j - x_i + v_{ij},
\end{equation}
with zero-mean Gaussian measurement noise $v_{ij} \in \mathcal{N}(\zero{3},R_{ij})$
and covariance matrix $R_{ij} = R_{ij}^\top \succ 0$.

Gradient:
\begin{align}
    f(\mathcal{D}) 
    &= \sum_{(i,j)\in E} \norm{x_j - x_i - \tilde{d}_{ij}}_{R_{ij}^{-1}}^2
    \\
    &= \sum_{(i,j)\in E} (x_j - x_i - \tilde{d}_{ij})^\top R_{ij}^{-1} (x_j - x_i - \tilde{d}_{ij}).
\end{align}

The gradient of $f$ with respect to $x_i\in\mathcal{M}_i$ is given by
\begin{equation}
    \grad_{x_i} f(x) = - 2 \sum_{j : (i,j)\in E} R_{ij}^{-1} (x_j - x_i - \tilde{d}_{ij}) 
    + 2 \sum_{j : (j,i) \in E} R_{ji}^{-1} (x_i - x_j - \tilde{d}_{ji}).
\end{equation}


\subsection{Rotation Averaging}
Manifold: Unit Quaternion,
\begin{equation}
    \mathcal{M}_i = S^3.
\end{equation}

Objective function:
\begin{equation}
    f(x) = \sum_{(i,j)\in E} \norm{R(x_i)^\top R(x_j) - \tilde{R}_{ij}}_F^2,
\end{equation}
where $E\in\seq{N}^2$ is a set of directed measurement edges and $\tilde{R}_{ij}$ is a noisy measurement given by
\begin{equation}
    \tilde{R}_{ij} = R(x_i)^\top R(x_j) V_{ij},
\end{equation}
where $V_{ij}$ is a perturbation rotation matrix corresponding to the rotation about a random axis by a zero-mean Gaussian angle.

Gradient:
Let $\bar{f}$ be the smooth extension of $f$ to $\R{4}$, given by
\begin{equation}
    \bar{f}(x) = \sum_{(i,j)\in E} \norm{R(x_i)^\top R(x_j) - \tilde{R}_{ij}}_F^2.
\end{equation}

Define $R_i, R_j \in SO(3)$ via
\begin{align}
    R_i &= R(x_i)
    \\
    R_j &= R(x_j).
\end{align}

Plug in $R_i$ for all $i\in\seq{N}$ into $\bar{f}$.
Note that for all $i\in\seq{N}$, the variable $x_i$ only appears through $R_i$.
As such, we define $\tilde{f}$ in terms of $R_i$ via
\begin{equation}
    \tilde{f}(\{R_1,\dots,R_N\}) = \sum_{(i,j)\in E} \norm{R_i^\top R_j - \tilde{R}_{ij}}_F^2.
\end{equation}

Choose $i\in\seq{N}$ and choose a matrix $V_i \in \R{3\times 3}$.
Define $V = \{\zero{},\dots,V_i,\dots,\zero{}\}$
Then,
\begin{align}
    D \tilde{f}(\{R_1,\dots,R_N\})[V]
    &= \lim_{t\to0} \frac{\tilde{f}(\{R_1,\dots,R_i+tV_i,\dots,R_N\}) - \tilde{f}(\{R_1,\dots,R_N\})}{t}
    \\
    &= \lim_{t\to0} \frac{1}{t} \Bigg( \sum_{j:(i,j)\in E} \left( \norm{(R_i+tV_i)^\top R_j - \tilde{R}_{ij}}_F^2 - \norm{R_i^\top R_j - \tilde{R}_{ij}}_F^2 \right) \\ 
    &\qquad + \sum_{j:(j,i)\in E} \left( \norm{R_j^\top (R_i+tV_i) - \tilde{R}_{ji}}_F^2 - \norm{R_j^\top R_i - \tilde{R}_{ji}}_F^2\right) \Bigg)
    \\
    &= \lim_{t\to0} \frac{1}{t} \Bigg( \sum_{j:(i,j)\in E} \left( \norm{R_i^\top R_j - \tilde{R}_{ij} + tV_i^\top R_j}_F^2 - \norm{R_i^\top R_j - \tilde{R}_{ij}}_F^2 \right) \\ 
    &\qquad + \sum_{j:(j,i)\in E} \left( \norm{R_j^\top R_i - \tilde{R}_{ji} + t R_j^\top V_i}_F^2 - \norm{R_j^\top R_i - \tilde{R}_{ji}}_F^2\right) \Bigg)
    \\
    &= \lim_{t\to0} \frac{1}{t} \Bigg( \sum_{j:(i,j)\in E} \trace \left( 2 \left(tV_i^\top R_j\right)^\top \left(R_i^\top R_j - \tilde{R}_{ij}\right) + \left(tV_i^\top R_j\right)^\top \left(tV_i^\top R_j\right) \right) \\ 
    &\qquad + \sum_{j:(j,i)\in E} \trace \left( 2 \left(t R_j^\top V_i\right)^\top \left(R_j^\top R_i - \tilde{R}_{ji}\right) + \left(t R_j^\top V_i\right)^\top \left(t R_j^\top V_i\right) \right) \Bigg)
    \\
    &= \lim_{t\to0} \frac{1}{t} \Bigg( \sum_{j:(i,j)\in E} \trace \left( 2 t R_j^\top V_i R_i^\top R_j - 2 t R_j^\top V_i \tilde{R}_{ij} + t^2 R_j^\top V_i V_i^\top R_j \right) \\ 
    &\qquad + \sum_{j:(j,i)\in E} \trace \left( 2 t V_i^\top R_j R_j^\top R_i - 2 t V_i^\top R_j \tilde{R}_{ji} + t^2 V_i^\top R_j R_j^\top V_i \right) \Bigg)
    \\
    &= \lim_{t\to0} \frac{1}{t} \Bigg( \sum_{j:(i,j)\in E} \trace \left( 2 t V_i R_i^\top - 2 t R_j^\top V_i \tilde{R}_{ij} + t^2 V_i V_i^\top \right) \\ 
    &\qquad + \sum_{j:(j,i)\in E} \trace \left( 2 t V_i^\top R_i - 2 t V_i^\top R_j \tilde{R}_{ji} + t^2 V_i^\top V_i \right) \Bigg)
    \\
    &= \lim_{t\to0} \Bigg( \sum_{j:(i,j)\in E} \trace \left( 2 V_i R_i^\top - 2 R_j^\top V_i \tilde{R}_{ij} + t V_i V_i^\top \right) \\ 
    &\qquad + \sum_{j:(j,i)\in E} \trace \left( 2 V_i^\top R_i - 2 V_i^\top R_j \tilde{R}_{ji} + t V_i^\top V_i \right) \Bigg)
    \\
    &= \sum_{j:(i,j)\in E} \trace \left( 2 V_i R_i^\top - 2 R_j^\top V_i \tilde{R}_{ij} \right) + \sum_{j:(j,i)\in E} \trace \left( 2 V_i^\top R_i - 2 V_i^\top R_j \tilde{R}_{ji} \right)
    \\
    &= \trace \left( \sum_{j:(i,j)\in E} \left( 2 V_i R_i^\top - 2 R_j^\top V_i \tilde{R}_{ij} \right) + \sum_{j:(j,i)\in E} \left( 2 V_i^\top R_i - 2 V_i^\top R_j \tilde{R}_{ji} \right) \right)
    \\
    &= \trace \left( \sum_{j:(i,j)\in E} \left( 2 V_i^\top R_i - 2 V_i^\top R_j \tilde{R}_{ij}^\top \right) + \sum_{j:(j,i)\in E} \left( 2 V_i^\top R_i - 2 V_i^\top R_j \tilde{R}_{ji} \right) \right)
    \\
    &= \trace \left(V_i^\top \left( \sum_{j:(i,j)\in E} \left( 2 R_i - 2 R_j \tilde{R}_{ij}^\top \right) + \sum_{j:(j,i)\in E} \left( 2 R_i - 2 R_j \tilde{R}_{ji} \right) \right) \right)
    \\
    &= \trace\left( V_i^\top \nabla_{R_i} \tilde{f}\right)
\end{align}

Thus, we know
\begin{equation}
    \nabla_{R_i} \tilde{f}(\{R_1,\dots,R_N\}) = \sum_{j:(i,j)\in E} \left( 2 R_i - 2 R_j \tilde{R}_{ij}^\top \right) + \sum_{j:(j,i)\in E} \left( 2 R_i - 2 R_j \tilde{R}_{ji} \right).
\end{equation}

Equivalently, we know
\begin{equation}
    \nabla_{R(x_i)} \bar{f}(x) = \sum_{j:(i,j)\in E} \left( 2 R(x_i) - 2 R(x_j) \tilde{R}_{ij}^\top \right) + \sum_{j:(j,i)\in E} \left( 2 R(x_i) - 2 R(x_j) \tilde{R}_{ji} \right).
\end{equation}

Finally, we again use chain rule, according to
\begin{equation}
    \nabla_{x_i} \bar{f}(x) = \begin{bmatrix}
        \langle \nabla_{R(x_i)} \bar{f}, \frac{\partial R(x_i)}{\partial q_w} \rangle_F
        \\
        \langle \nabla_{R(x_i)} \bar{f}, \frac{\partial R(x_i)}{\partial q_x} \rangle_F
        \\
        \langle \nabla_{R(x_i)} \bar{f}, \frac{\partial R(x_i)}{\partial q_y} \rangle_F
        \\
        \langle \nabla_{R(x_i)} \bar{f}, \frac{\partial R(x_i)}{\partial q_z} \rangle_F
    \end{bmatrix}
\end{equation}
with
\begin{equation}
    \grad_{x_i} f = P_{x_i} (\nabla_{x_i} \bar{f}).
\end{equation}


\subsection{Rotation-Invariant Formation Control}
Agent $i$'s manifold is given by
\begin{equation}
    \mathcal{M}_i = \R{3} \times S^3.
\end{equation}
Relevant variables for agent $i$:
\begin{itemize}
    \item Decision variable: $x_i = (t_i, q_i) \in (\R{3} \times S^3) \in \mathcal{M}_i$
    \item Prior position: $\bar{t}_i \in \R{3}$
    \item Prior orientation: $\bar{R}_i \in SO(3)$
    \item Relative offset to agent $j$ in local reference frame: $\tilde{t}_{ij}$
\end{itemize}

Objective function:
\begin{equation}
    f(x) = \alpha \sum_{i=1}^N \left( \norm{t_i - \bar{t}_i}_2^2 + \norm{R(q_i) - \bar{R}_i}_F^2 \right) + (1-\alpha) \frac{1}{N} \sum_{(i,j)\in E} \norm{R(q_i)^\top (t_j - t_i) - \tilde{t}_{ij}}_2^2.
\end{equation}

Gradient with respect to $t_i$:
\begin{align}
    f(x) 
    &= \alpha \sum_{i=1}^N \left( \norm{t_i - \bar{t}_i}_2^2 + \norm{R(q_i) - \bar{R}_i}_F^2 \right) + (1-\alpha) \frac{1}{N} \sum_{(i,j)\in E} \norm{R(q_i)^\top (t_j - t_i) - \tilde{t}_{ij}}_2^2
    \\ 
    &= \alpha \sum_{i=1}^N \left( t_i^\top t_i - 2t_i^\top \bar{t}_i + \bar{t}_i^\top \bar{t}_i + \norm{R(q_i) - \bar{R}_i}_F^2 \right) \\
    &\qquad + (1-\alpha) \frac{1}{N} \sum_{(i,j)\in E} \left( (t_j-t_i)^\top R(q_i) R(q_i)^\top (t_j-t_i) - 2 (t_j-t_i)^\top R(q_i) \tilde{t}_{ij} + \tilde{t}_{ij}^\top \tilde{t}_{ij} \right) 
    \\ 
    &= \alpha \sum_{i=1}^N \left( t_i^\top t_i - 2t_i^\top \bar{t}_i + \bar{t}_i^\top \bar{t}_i + \norm{R(q_i) - \bar{R}_i}_F^2 \right) \\
    &\qquad + (1-\alpha) \frac{1}{N} \sum_{(i,j)\in E} \left( (t_j-t_i)^\top (t_j-t_i) - 2 (t_j-t_i)^\top R(q_i) \tilde{t}_{ij} + \tilde{t}_{ij}^\top \tilde{t}_{ij} \right) 
    \\ 
    &= \alpha \sum_{i=1}^N \left( t_i^\top t_i - 2t_i^\top \bar{t}_i + \bar{t}_i^\top \bar{t}_i + \norm{R(q_i) - \bar{R}_i}_F^2 \right) \\
    &\qquad + (1-\alpha) \frac{1}{N} \sum_{(i,j)\in E} \left( t_j^\top t_j - 2t_i^\top t_j + t_i^\top t_i - 2 (t_j-t_i)^\top R(q_i) \tilde{t}_{ij} + \tilde{t}_{ij}^\top \tilde{t}_{ij} \right) 
\end{align}
\begin{align}
    \grad_{t_i} f(x)
    &= 2 \alpha (t_i - \bar{t}_i) 
    + 2(1-\alpha) \frac{1}{N} \sum_{j:(i,j)\in E} (t_i - t_j + R(q_i)\tilde{t}_{ij})
    + 2(1-\alpha) \frac{1}{N} \sum_{j:(j,i)\in E} (t_i - t_j - R(q_j)\tilde{t}_{ji})
\end{align}

Gradient with respect to $q_i$:

First, define $\bar{f}$ as smooth extension of $f$, where $\mathcal{M}_i$ is extended to $(\R{3} \times \R{4})$.
Additionally, define $R_i = R(q_i)$ and define $\tilde{f}$ as $\bar{f}$ with all instances of $R(q_i)$ replaced with $R_i$, given by
\begin{equation}
    \tilde{f}(\{t_i, R_i\}_{i\in\seq{N}}) = \alpha \sum_{i=1}^N \left( \norm{t_i - \bar{t}_i}_2^2 + \norm{R_i - \bar{R}_i}_F^2 \right) + (1-\alpha) \frac{1}{N} \sum_{(i,j)\in E} \norm{R_i^\top (t_j - t_i) - \tilde{t}_{ij}}_2^2.
\end{equation}
Choose a matrix $V_i\in\R{3\times 3}$, and define $V = \{(\zero{3},\zero{3\times 3})^{i-1},(\zero{},V_i),(\zero{3},\zero{3\times 3})^{N-i}\}$.
Then,
\begin{align}
    D\tilde{f}(\{t_i, R_i\}_{i\in\seq{N}})[V]
    &= \lim_{t\to0} \frac{\tilde{f}((t_1,R_1),\dots,(t_i,R_i+tV_i),\dots,(t_N,R_N))-\tilde{f}(\{t_i, R_i\}_{i\in\seq{N}})}{t}
    \\
    &= \lim_{t\to0} \frac{1}{t} \bigg( \alpha \norm{R_i+tV_i-\bar{R}_i}_F^2 - \alpha \norm{R_i-\bar{R}_i}_F^2 \\
    &\qquad + (1-\alpha) \frac{1}{N} \sum_{j:(i,j)\in E} \left( \norm{(R_i+tV_i)^\top (t_j-t_i) - \tilde{t}_{ij}}_2^2 - \norm{R_i^\top (t_j-t_i) - \tilde{t}_{ij}}_2^2 \right) \bigg)
    \\
    &= \lim_{t\to0} \frac{1}{t} \bigg( \alpha \norm{R_i-\bar{R}_i+tV_i}_F^2 - \alpha \norm{R_i-\bar{R}_i}_F^2 \\
    &\qquad + (1-\alpha) \frac{1}{N} \sum_{j:(i,j)\in E} \left( \norm{R_i^\top (t_j-t_i) - \tilde{t}_{ij} + tV_i^\top (t_j-t_i)}_2^2 - \norm{R_i^\top (t_j-t_i) - \tilde{t}_{ij}}_2^2 \right) \bigg)
    \\
    &= \lim_{t\to0} \frac{1}{t} \bigg( \alpha \trace\left( 2tV_i^\top (R_i-\bar{R}_i) + t^2V_i^\top V_i\right) \\
    &\qquad + (1-\alpha) \frac{1}{N} \sum_{j:(i,j)\in E} \left( 2t(R_i^\top(t_j-t_i)-\tilde{t}_{ij})^\top V_i^\top(t_j-t_i) + t^2 (t_j-t_i)^\top V_i V_i^\top (t_j-t_i) \right) \bigg)
    \\
    &= \lim_{t\to0} \bigg( \alpha \trace\left( 2V_i^\top (R_i-\bar{R}_i) + tV_i^\top V_i\right) \\
    &\qquad + (1-\alpha) \frac{1}{N} \sum_{j:(i,j)\in E} \left( 2(R_i^\top(t_j-t_i)-\tilde{t}_{ij})^\top V_i^\top(t_j-t_i) + t (t_j-t_i)^\top V_i V_i^\top (t_j-t_i) \right) \bigg)
    \\
    &= \alpha \trace\left( 2V_i^\top (R_i-\bar{R}_i)\right) + (1-\alpha) \frac{1}{N} \sum_{j:(i,j)\in E} 2(R_i^\top(t_j-t_i)-\tilde{t}_{ij})^\top V_i^\top(t_j-t_i)
    \\
    &= \alpha \trace\left( 2V_i^\top (R_i-\bar{R}_i)\right) + (1-\alpha) \frac{1}{N} \trace\left( \sum_{j:(i,j)\in E} 2(R_i^\top(t_j-t_i)-\tilde{t}_{ij})^\top V_i^\top(t_j-t_i) \right)
    \\
    &= \alpha \trace\left( 2V_i^\top (R_i-\bar{R}_i)\right) + (1-\alpha) \frac{1}{N} \trace\left( \sum_{j:(i,j)\in E} 2 V_i^\top(t_j-t_i) (R_i^\top(t_j-t_i)-\tilde{t}_{ij})^\top \right)
    \\
    &= \trace\left( 2\alpha V_i^\top (R_i-\bar{R}_i) + (1-\alpha) \frac{1}{N} \sum_{j:(i,j)\in E} 2 V_i^\top(t_j-t_i) (R_i^\top(t_j-t_i)-\tilde{t}_{ij})^\top \right)
    \\
    &= \trace\left( V_i^\top \left( 2\alpha (R_i-\bar{R}_i) + (1-\alpha) \frac{1}{N} \sum_{j:(i,j)\in E} 2 (t_j-t_i) (R_i^\top(t_j-t_i)-\tilde{t}_{ij})^\top \right) \right)
    \\
    &= \trace\left( V_i^\top \nabla_{R_i} \tilde{f} \right)
\end{align}

Thus, we know
\begin{equation}
    \nabla_{R_i} \tilde{f}(\{t_i, R_i\}_{i\in\seq{N}})
    = 2\alpha (R_i-\bar{R}_i) + (1-\alpha) \frac{1}{N} \sum_{j:(i,j)\in E} 2 (t_j-t_i) (R_i^\top(t_j-t_i)-\tilde{t}_{ij})^\top.
\end{equation}

Equivalently, we know
\begin{equation}
    \nabla_{R(q_i)} \bar{f}(x)
    = 2\alpha (R(q_i)-\bar{R}_i) + (1-\alpha) \frac{1}{N} \sum_{j:(i,j)\in E} 2 (t_j-t_i) (R(q_i)^\top(t_j-t_i)-\tilde{t}_{ij})^\top.
\end{equation}

Finally, we again use chain rule, according to
\begin{equation}
    \nabla_{q_i} \bar{f}(x) = \begin{bmatrix}
        \langle \nabla_{R(q_i)} \bar{f}, \frac{\partial R(q_i)}{\partial q_{iw}} \rangle_F
        \\
        \langle \nabla_{R(q_i)} \bar{f}, \frac{\partial R(q_i)}{\partial q_{ix}} \rangle_F
        \\
        \langle \nabla_{R(q_i)} \bar{f}, \frac{\partial R(q_i)}{\partial q_{iy}} \rangle_F
        \\
        \langle \nabla_{R(q_i)} \bar{f}, \frac{\partial R(q_i)}{\partial q_{iz}} \rangle_F
    \end{bmatrix}
\end{equation}
with
\begin{equation}
    \grad_{q_i} f = P_{q_i} (\nabla_{q_i} \bar{f}).
\end{equation}


\subsection{Dictionary Learning}
Setup:
\begin{itemize}
    \item Let $y_1,\dots,y_k \in \R{d}$ be a collection of $k$ images of dimension $d$
    \item Let $b_1,\dots,b_n \in \R{d}$ be a dictionary of $n$ atoms of dimension $d$
    \item Let $Y = \begin{bmatrix}
        y_1 & \cdots & y_k
    \end{bmatrix} \in \R{d\times k}$ be the ``data matrix''
    \item Let $D = \begin{bmatrix}
        b_1 & \cdots & b_n
    \end{bmatrix} \in\R{d\times n}$ be the ``dictionary matrix''
    \item We normalize $b_i$ such that $\norm{b_i}_2^2 = 1$, meaning $b_i\in S^{d-1}$ and $D\in(S^{d-1})^n$
    \item Let $C\in\R{n\times k}$ be the ``coefficient matrix''
    \item We seek to find $D\in(S^{d-1})^n$ and $C\in\R{n\times k}$ such that $Y\approx DC$ and $C$ is sparse 
    \item For sparsity of $C$, we allow $n > d$
    \item For dictionary compression, we want $n << k$
\end{itemize}

Discontinuous problem:
\begin{equation}
    \min_{D\in(S^{d-1})^n, C\in\R{n\times k}} f(D,C) 
    = \frac{1}{k} \norm{Y-DC}_F^2 + \frac{1}{k} \lambda \norm{C}_0
\end{equation}

Continuous, non-smooth problem:
\begin{align}
    \min_{D\in(S^{d-1})^n, C\in\R{n\times k}} f(D,C)
    &= \frac{1}{k} \norm{Y-DC}_F^2 + \frac{1}{k} \lambda \norm{C}_1
    \\
    &= \frac{1}{k} \norm{Y-DC}_F^2 + \frac{1}{k} \lambda \sum_{j=1}^n \sum_{\ell=1}^k |C_{j\ell}|
\end{align}

Smooth problem:
\begin{align}
    \min_{D\in(S^{d-1})^n, C\in\R{n\times k}} f(D,C) 
    = \frac{1}{k} \norm{Y-DC}_F^2 + \frac{1}{k} \lambda \sum_{j=1}^n \sum_{\ell=1}^k \sqrt{C_{j\ell}^2 + \varepsilon}
\end{align}

We split $n$ into $N$ partitions $n_1,\dots,n_N$ such that $\sum_{i=1}^N n_i = n$.
Agent $i$'s decision variables will be 
$D_i\in (S^{d-1})^{n_i}$ and $C_i\in\R{n_i\times k}$ such that
\begin{equation}
    D = \begin{bmatrix}
        D_1 & \cdots & D_N
    \end{bmatrix},
    \qquad\text{and}\qquad
    C = \begin{bmatrix}
        C_1 \\ \vdots \\ C_N
    \end{bmatrix}.
\end{equation}
Agent $i$'s manifold is the product manifold
\begin{equation}
    \mathcal{M}_i = (S^{d-1})^{n_i} \times \R{n_i\times k}.
\end{equation}

The problem can be restated as
\begin{align}
    \min_{D_i\in(S^{d-1})^{n_i}, C_i\in\R{n_i\times k}} f\left(\{(D_i,C_i)\}_{i\in\seq{N}}\right)
    &= \frac{1}{k} \norm{Y - \sum_{i=1}^N D_i C_i}_F^2 + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
\end{align}

Let $D_{ij}\in S^{d-1}$ be the \superth{j}-column of $D_i$ and let $C_{ij}\in\R{1\times k}$ be the \superth{j}-row of $C_i$, such that
\begin{equation}
    D_i=\begin{bmatrix}
        D_{i1} & \cdots & D_{in_i}
    \end{bmatrix},
\qquad\text{and}\qquad
    C_i = \begin{bmatrix}
        C_{i1} \\ \vdots \\ C_{in_i}
    \end{bmatrix}.
\end{equation}

The objective function can be restated as
\begin{align}
    f\left(\{(D_i,C_i)\}_{i\in\seq{N}}\right)
    &= \frac{1}{k} \norm{Y - \sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij} C_{ij}}_F^2 + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
\end{align}

Gradient with respect to $D_{IJ}$ for $I\in\seq{N}$ and $J\in\seq{n_I}$:

Objective
\begin{align}
    &f\left(\{(D_i,C_i)\}_{i\in\seq{N}}\right)
    \\
    &= \frac{1}{k} \norm{Y - \sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij} C_{ij}}_F^2 + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
    \\
    &= \frac{1}{k} \trace\left(\left(Y - \sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij} C_{ij}\right)^\top\left(Y - \sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij} C_{ij}\right)\right) + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
    \\
    &= \frac{1}{k} \trace\left( Y^\top Y - 2Y^\top \sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij} C_{ij} + \left(\sum_{i=1}^N \sum_{j=1}^{n_i} C_{ij}^\top D_{ij}^\top \right) \left(\sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij} C_{ij}\right) \right) \\
    &\qquad + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
    \\
    &= \frac{1}{k} \trace\left( Y^\top Y \right)
    - 2\frac{1}{k} \trace\left( Y^\top \sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij} C_{ij} \right)
    + \frac{1}{k} \trace\left( \left(\sum_{i_1=1}^N \sum_{j_1=1}^{n_{i_1}} C_{i_1j_1}^\top D_{i_1j_1}^\top \right) \left(\sum_{i_2=1}^N \sum_{j_2=1}^{n_{i_2}} D_{i_2j_2} C_{i_2j_2}\right) \right) \\
    &\qquad + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
    \\
    &= \frac{1}{k} \trace\left( Y^\top Y \right)
    - 2\frac{1}{k} \trace\left( \sum_{i=1}^N \sum_{j=1}^{n_i} Y^\top D_{ij} C_{ij} \right)
    + \frac{1}{k} \trace\left( \sum_{i_1=1}^N \sum_{j_1=1}^{n_{i_1}} \sum_{i_2=1}^N \sum_{j_2=1}^{n_{i_2}} C_{i_1j_1}^\top D_{i_1j_1}^\top D_{i_2j_2} C_{i_2j_2} \right) \\
    &\qquad + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
    \\
    &= \frac{1}{k} \trace\left( Y^\top Y \right)
    - 2\frac{1}{k} \trace\left( \sum_{i=1}^N \sum_{j=1}^{n_i} C_{ij} Y^\top D_{ij} \right)
    + \frac{1}{k} \trace\left( \sum_{i_1=1}^N \sum_{j_1=1}^{n_{i_1}} \sum_{i_2=1}^N \sum_{j_2=1}^{n_{i_2}} D_{i_1j_1}^\top D_{i_2j_2} C_{i_2j_2} C_{i_1j_1}^\top \right) \\
    &\qquad + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
    \\
    &= \frac{1}{k} \trace\left( Y^\top Y \right)
    - 2\frac{1}{k} \trace\left( \sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij}^\top Y C_{ij}^\top \right)
    + \frac{1}{k} \trace\left( \sum_{i_1=1}^N \sum_{j_1=1}^{n_{i_1}} \sum_{i_2=1}^N \sum_{j_2=1}^{n_{i_2}} D_{i_1j_1}^\top D_{i_2j_2} C_{i_2j_2} C_{i_1j_1}^\top \right) \\
    &\qquad + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
    \\
    &= \frac{1}{k} \trace\left( Y^\top Y \right)
    - 2 \frac{1}{k} \sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij}^\top Y C_{ij}^\top
    + \frac{1}{k} \sum_{i_1=1}^N \sum_{j_1=1}^{n_{i_1}} \sum_{i_2=1}^N \sum_{j_2=1}^{n_{i_2}} D_{i_1j_1}^\top D_{i_2j_2} C_{i_2j_2} C_{i_1j_1}^\top \\
    &\qquad + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
\end{align}

Euclidean Gradient:
\begin{align}
    \nabla_{D_{IJ}} f\left(\{(D_i,C_i)\}_{i\in\seq{N}}\right)
    &= -2 \frac{1}{k} Y C_{IJ}^\top
    + \frac{1}{k} \sum_{i_2=1}^N \sum_{j_2=1}^{n_{i_2}} D_{i_2j_2} C_{i_2j_2} C_{IJ}^\top
    + \frac{1}{k} \sum_{i_1=1}^N \sum_{j_1=1}^{n_{i_1}} D_{i_1j_1} C_{IJ} C_{i_1j_1}^\top
    \\
    &= -2 \frac{1}{k} Y C_{IJ}^\top
    + \frac{1}{k} \sum_{i_2=1}^N \sum_{j_2=1}^{n_{i_2}} D_{i_2j_2} C_{i_2j_2} C_{IJ}^\top
    + \frac{1}{k} \sum_{i_1=1}^N \sum_{j_1=1}^{n_{i_1}} D_{i_1j_1} C_{i_1j_1} C_{IJ}^\top
    \\
    &= -2 \frac{1}{k} Y C_{IJ}^\top
    + 2 \frac{1}{k} \sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij} C_{ij} C_{IJ}^\top
    \\
    &= -2 \frac{1}{k} Y C_{IJ}^\top
    + 2 \frac{1}{k} DC C_{IJ}^\top
    \\
    &= -2 \frac{1}{k} (Y - DC) C_{IJ}^\top
    \\
    \nabla_{D_I} f\left(\{(D_i,C_i)\}_{i\in\seq{N}}\right)
    &= -2 \frac{1}{k} (Y - DC) C_{I}^\top
\end{align}

Gradient with respect to $(C_I)_{JL}$ for $I\in\seq{N}$, $J\in\seq{n_I}$, and $L\in\seq{k}$:

Objective:
\begin{align}
    f\left(\{(D_i,C_i)\}_{i\in\seq{N}}\right)
    &= \frac{1}{k} \trace\left( Y^\top Y \right)
    - 2 \frac{1}{k} \sum_{i=1}^N \sum_{j=1}^{n_i} D_{ij}^\top Y C_{ij}^\top
    + \frac{1}{k} \sum_{i_1=1}^N \sum_{j_1=1}^{n_{i_1}} \sum_{i_2=1}^N \sum_{j_2=1}^{n_{i_2}} D_{i_1j_1}^\top D_{i_2j_2} C_{i_2j_2} C_{i_1j_1}^\top \\

    \\

    - 2 \frac{1}{k} \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k (D_{ij}^\top Y)_{1\ell} (C_i)_{j\ell}
    + \frac{1}{k} \sum_{i_1=1}^N \sum_{j_1=1}^{n_{i_1}} \sum_{i_2=1}^N \sum_{j_2=1}^{n_{i_2}} \sum_{\ell=1}^k D_{i_1j_1}^\top D_{i_2j_2} (C_{i_2})_{j_2\ell} (C_{i_1})_{j_1\ell} \\
    &\qquad + \frac{1}{k} \lambda \sum_{i=1}^N \sum_{j=1}^{n_i} \sum_{\ell=1}^k \sqrt{(C_i)_{j\ell}^2 + \varepsilon}
\end{align}

\begin{align}
    \nabla_{(C_I)_{JL}} f\left(\{(D_i,C_i)\}_{i\in\seq{N}}\right)
    &= -2 \frac{1}{k} (D_{IJ}^\top Y)_{1L} + 2\sum_{i=1}^N \sum_{j=1}^{n_i} D_{IJ}^\top D_{ij} (C_i)_{jL}

    &= -2 \frac{1}{k} D_I^\top (Y - DC) + \frac{1}{k} \lambda C_I \oslash \sqrt{C_I \circ C_I + \mathbf{1}^{n_I\times k} \varepsilon},
\end{align}
where $\oslash$, $\circ$, and $\sqrt{\cdot}$ respectively denote the element-wise division, multiplication, and square root operators.


\subsubsection{Lipschitz Continuity}

For a square matrix $A\in\R{n\times n}$, we define the functions 
$\diag:\R{n\times n}\to\R{n}$ and $\ddiag:\R{n\times n}\to\R{n\times n}$ via
\begin{align}
    \diag(A) &= \begin{bmatrix}
        A_{11} \\ A_{22} \\ \vdots \\ A_{nn}
    \end{bmatrix},
    \\
    \ddiag(A) &= \begin{bmatrix}
        A_{11} & 0 & \cdots & 0 \\
        0 & A_{22} & \cdots & 0 \\
        \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & \cdots & A_{nn}
    \end{bmatrix}.
\end{align}

Oblique manifold:

\begin{equation}
    \text{OB}(d,n) = \{X\in\R{d\times n} : \diag(X^\top X) = \mathbf{1}\}.
\end{equation}

Projection:
For $X\in\text{OB}(d,n)$ and $V\in\R{d\times n}$
\begin{equation}
    P_X(V) = V - X \ddiag(X^\top V)
\end{equation}
